+const CellDataTypeMapping: Record<string, PowerBIDataType> = {
+    number: "Number",
+    boolean: "Logical",
+    date: "DateTime",
+    dateString: "DateTime",
+    dateTime: "DateTime",
+    dateTimeString: "DateTime",

+const IsDate = (value: unknown): value is Date => value instanceof Date && !isNaN(value.getTime());
+
+const InferDataType = (values: unknown[]): PowerBIDataType => {
+    const value = values.find(item => item !== null && item !== undefined && item !== "");
+    if (typeof value === "number") return "Number";
+    if (typeof value === "boolean") return "Logical";
+    if (IsDate(value)) return "DateTime";
+    return "Text";
+};
+
+// Values are parsed by Power BI using M language value specification
+const FormatValue = (value: unknown, formattedValue: string | null, dataType: PowerBIDataType): string => {
+    if (value === null || value === undefined) return "";
+    switch (dataType) {
+        case "Number":
+            return typeof value === "number" && isFinite(value) ? String(value) : "";
+        case "Logical":
+            return typeof value === "boolean" ? String(value) : "";
+        case "DateTime":
+            if (IsDate(value)) return value.toISOString();
+            return typeof value === "string" ? value : "";
+        default:
+            if (typeof value === "object") return formattedValue ?? JSON.stringify(value);
+            return String(value);
+    }
+};
+
+const GetUniqueName = (name: string, usedNames: Set<string>): string => {
+    let uniqueName = name || "Column";
+    let index = 2;
+    while (usedNames.has(uniqueName)) {
+        uniqueName = `${name} (${index++})`;
+    }
+    usedNames.add(uniqueName);
+    return uniqueName;
+};

+
+export const PowerBIReportOverlay: React.FC<IPowerBIReportOverlayProps> = ({ request, onClose }) => {
+    const containerRef = useRef<HTMLDivElement>(null);
+    const [isLoading, setIsLoading] = useState<boolean>(true);
+
+    useEffect(() => {
+        const container = containerRef.current;
+        const powerBI = Application.Instance.DataFacade.PowerBI;
+        let isActive = true;
+

+            powerBI.ResetContainer(container);
+        };

+

+        <Overlay

+            width="90vw"
+            centered

+            maskClosable={false}
+            footer={null}

+                <div className="Advantage-DataGrid-PowerBI-Loader">
+                    <Loader size={LoaderSize.M} />
+                </div>

+      const accessToken = await this.GetAccessToken();
+      const locale = request.locale || "en_US";
+
+      const config: PowerBIModels.IQuickCreateConfiguration = {
+        type: "quickCreate",
+        tokenType: models.TokenType.Aad,
+        accessToken,
+        embedUrl: PowerBIBridge.QuickCreateEmbedUrl,
+        datasetCreateConfig: {
+          locale,
+          tableSchemaList: [{
+            // Only single table named "Table" is supported by quick create
+            name: "Table",
+            columns: request.columns.map(column => ({
+              name: column.name,
+              dataType: column.dataType as PowerBIModels.DataType,
+            })),
+          }],
+          data: [{ name: "Table", rows: request.rows }],
+        },
+        reportCreationMode: "QuickExplore" as PowerBIModels.ReportCreationMode,
+        settings: {
+          localeSettings: { language: locale },
+        },
+      };
+
+      const requestSize = new Blob([JSON.stringify(config.datasetCreateConfig)]).size;
+      if (requestSize > PowerBIBridge.MaxRequestSize) {
+        throw new Error(`Power BI: data size (${requestSize} bytes) exceeds quick create limit of 16MB`);
+      }
+
+      powerBIService.reset(container);
+      this.SetReportState("creating", null);
+
+      const report = powerBIService.quickCreate(container, config);
+      report.on("loaded", () => this.SetReportState("loaded", report));
+      report.on("rendered", () => this.SetReportState("rendered", report));
+      report.on("error", (event: any) => {
+        this.ConsoleMessage(`Power BI: report error ${JSON.stringify(event?.detail)}`, true);
+        this.SetReportState("error", report, event?.detail);
+      });
+      report.on("dataSelected", (event: any) => {
+        const reportEvent: IPowerBIReportEvent = { report, detail: event?.detail };
+        this.NotifyAll(this.eventNames.DataSelected, reportEvent);
+      });
+
+      this.ConsoleMessage(
+        `Power BI: Quick create report '${request.modelName}' with ${request.rows.length} rows`
+      );
+      return report;
+    } catch (error) {
+      this.ConsoleError(error as Error);
+      this.ConsoleMessage("Power BI: Failed to create report", true);
+      this.SetReportState("error", null, error);
+      throw error;
+    }
+  }
+
+  /**
+   * Remove embedded report from the container
+   */
+  public ResetContainer(container: HTMLElement): void {
+    if (this.powerBIService && container) {
+      this.powerBIService.reset(container);
+    }
+  }
+
+  public Dispose(): void {
+    this.powerBIService = null;
+    this.ConsoleMessage("Power BI: Disposed");
+  }
+
+  private async GetService(): Promise<PowerBIService.Service> {
+    if (!this.powerBIService) {
+      const { service, factories } = await import("powerbi-client");
+      this.powerBIService = new service.Service(
+        factories.hpmFactory,
+        factories.wpmpFactory,
+        factories.routerFactory
+      );
+    }
+    return this.powerBIService;
+  }
+
+  /**
+   * Power BI quick create supports Microsoft Entra (AAD) tokens only
+   */
+  private GetAccessToken(): Promise<string> {
+    return new Promise<string>((resolve, reject) => {
+      // AppendAuthorization doesn't call back when token can't be acquired
+      const timeout = setTimeout(
+        () => reject(new Error("Power BI: Failed to acquire access token")),
+        PowerBIBridge.TokenTimeout
+      );
+      Application.Instance.Tools.GetApplicationSecurity("Azure").AppendAuthorization(
+        PowerBIBridge.ServiceName,
+        token => {
+          clearTimeout(timeout);
+          token ? resolve(token) : reject(new Error("Power BI: Failed to acquire access token"));
+        }
+      );
+    });
+  }
+
+  private SetReportState(state: PowerBIReportState, report: Embed | null, detail?: any): void {
+    const reportEvent: IPowerBIReportEvent = { report, detail };
+    this.NotifyAll(this.eventNames.ReportStateChange, state, reportEvent);
+  }
